sinodc_dec_develop_construction = { #工业化改造
	sinodc_test = {
		priority = 8
		visible = {
			is_debug = yes
		}
		complete_effect = {
			add_to_variable = { 
				variable_dcp = 10
				tooltip = sinodc_tip_variable_dcp_add
			}  
			add_to_variable = { 
				variable_dcp_max = 10
			} 
		}
		complete_effect = {  
			if = {
				limit = {
					check_variable = { SNL_var_days_test > 0 } #调试_决议时间1
				}
				clear_variable = SNL_var_days_test #调试_决议时间1
			}
			else = {
				set_variable = { SNL_var_days_test = 1 } #调试_决议时间1
			}
		}
	}

		}

	sinodc_dec_add_pp_start = { #开始增加政治点
		icon = generic_form_nation
		cost = 10
		fire_only_once = no 
		priority = 6
		ai_will_do = {
			factor = 0
		}
		available = {
			num_of_civilian_factories_available_for_projects > 1 #检查可用工厂数量
			custom_trigger_tooltip = {
				tooltip = sinodc_tip_dcp_2 #x/2
				check_variable = { variable_dcp > 1 } #检查改造点数量
			}
        }
		visible = {
			has_country_flag = sinodc_fla_develop_construction_start #开启决议的flag
			has_country_flag = sinodc_fla_decisions_show #决议显示
			not = { has_country_flag = sinodc_fla_add_pp_start } #开始增加pp
		}
		complete_effect = {
            add_to_variable = { 
				variable_dcp = -2
				tooltip = sinodc_tip_variable_dcp_occupy #减少改造点
			}    
			add_dynamic_modifier = { modifier = sinodc_dym_add_pp }
			set_country_flag = sinodc_fla_add_pp_start #开始增加pp
			activate_mission = sinodc_dec_add_pp #激活增加pp的任务
        }
	}

		icon = generic_form_nation
		fire_only_once = no

		priority = 5
		activation = {

		}                                    
		available = {   
			always = no                        
		}
		cancel_trigger = { #取消任务的条件
			not = { has_country_flag = sinodc_fla_add_pp_start } #开始增加pp
		}
		timeout_effect = { #过期时触发的效果
			add_political_power = 20
			activate_mission = sinodc_dec_add_pp #重新激活任务
		}
		cancel_effect = { #取消时的效果
			add_to_variable = { variable_dcp = 2 tooltip = sinodc_tip_variable_dcp_release }  
			remove_dynamic_modifier = { modifier = sinodc_dym_add_pp }
		}
